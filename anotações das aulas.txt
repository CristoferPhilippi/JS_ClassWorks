Desafio adicional — Evitar divisão por zero

Em uma calculadora, dividir por zero não deve ser permitido.

Crie:

function dividir(numero1, numero2) {
  // código
}

Quando numero2 for zero, retorne:

Não é possível dividir por zero

Caso contrário, devolva a divisão.

Testes:

console.log(dividir(10, 2));
console.log(dividir(10, 0));

Resultados:

5
Não é possível dividir por zero
Resumo da aula
Função    → bloco reutilizável de código
Parâmetro → variável recebida pela função
Argumento → valor enviado durante a execução
return    → devolve um resultado
console.log → apenas mostra um valor
Escopo    → região em que uma variável existe


//resultado desafio adicional

function dividir(numero1, numero2) {
  if (numero2 === 0) {
    return "Não é possível dividir por zero";
  } else {
    const divisao = numero1 / numero2;
    return divisao;
  }
}

console.log(dividir(10, 2));
console.log(dividir(10, 0));

<============= Comparação entre as repetições ============>
while

Use quando a repetição depende de uma condição e você não sabe exatamente quantas vezes acontecerá.

while (saldo > 0) {
}
do...while

Use quando o bloco precisa executar pelo menos uma vez.

do {
} while (condicao);
for

Use quando existe um contador ou você precisa controlar o índice.

for (let indice = 0; indice < nomes.length; indice++) {
}
for...of

Use quando deseja percorrer diretamente os valores de um array.

for (const nome of nomes) {
}

function calcularMedia(notas) {
  
let soma = 0;
  
  for (const nota of notas) {
    soma += nota;
       
  }
  const media = soma / notas.length;
  return media;

function cadastrarUsuario(nome) {

const usuarios = [];

function cadastrarUsuario(nome) {
  if (nome === '') {
    return "Nome inválido";
  }

  if (usuarios.includes(nome)) {
    return "Usuário já cadastrado";
  }

  usuarios.push(nome);

  return "Usuário cadastrado com sucesso";
}
